\begin{theorem}[Universal Spectral Floor]
\label{thm:spectral-floor}
\lean{HoggarBound.hoggar_bound, HoggarBound.NonVacuity.spectral_floor_nonvacuous}
\leanok
For every $N \ge 1$, every subset $S \subseteq W(2N-1,2)$ of non-identity $N$-qubit Pauli observables, and every $\pm 1$ assignment $f$, the context balance satisfies $(D-1)\beta_6(S,f) \le (D-2)|S|\sqrt{(D-1)|S|}$ where $D = 2^N$.
\end{theorem}
